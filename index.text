  <script>
    // ========== CONFIG — UPDATE AFTER DEPLOY ==========
    const MONAD_CHAIN_ID = 10143;
    const RPC_URL = "https://testnet-rpc.monad.xyz";
    const EXPLORER = "https://testnet.monadexplorer.com";

    // Placeholder addresses — replace with deployed ones
    const CONTRACTS = {
      registry: "0x0000000000000000000000000000000000000000",
      fanToken: "0x0000000000000000000000000000000000000000",
      tournamentNFT: "0x0000000000000000000000000000000000000000",
      predictionMarket: "0x0000000000000000000000000000000000000000",
      mockUSDC: "0x0000000000000000000000000000000000000000"
    };

    const OWNER_SECURITY_KEY = "fanarena-owner-2026";
    let isAdminUnlocked = false;
    let provider, signer, account;

    // Minimal ABIs
    const ERC20_ABI = [
      "function balanceOf(address) view returns (uint256)",
      "function transfer(address,uint256) returns (bool)",
      "function approve(address,uint256) returns (bool)",
      "function mint(address,uint256)",
      "function faucet()",
      "function decimals() view returns (uint8)"
    ];
    const NFT_ABI = [
      "function balanceOf(address) view returns (uint256)",
      "function mint(address)"
    ];
    const REGISTRY_ABI = [
      "function addCreator(string,string,string,string,address) returns (uint256)",
      "function addSwappableToken(address)",
      "function creatorCount() view returns (uint256)",
      "function getCreator(uint256) view returns (tuple(string name,string bio,string description,string remixProfile,address fanToken,bool active))",
      "function getSwappableTokens() view returns (address[])"
    ];
    const MARKET_ABI = [
      "function placeBet(uint8,uint256)",
      "function getMarketInfo() view returns (string,string,string,bool,uint8,uint256,uint256)",
      "function betsA(address) view returns (uint256)",
      "function betsB(address) view returns (uint256)"
    ];

    // Auto-connect attempt
    window.addEventListener('load', async () => {
      if (window.ethereum) {
        try {
          const accounts = await window.ethereum.request({ method: 'eth_accounts' });
          if (accounts.length > 0) {
            await connectWallet(true);
          }
        } catch (e) { console.log("Auto-connect skipped"); }
      }
      loadCreators();
      loadMarket();
    });

    document.getElementById('connectBtn').onclick = () => connectWallet(false);
    document.getElementById('disconnectBtn').onclick = disconnectWallet;

    async function connectWallet(silent = false) {
      if (!window.ethereum) {
        alert("Please install MetaMask or an EVM wallet.");
        return;
      }
      try {
        provider = new ethers.providers.Web3Provider(window.ethereum);
        if (!silent) {
          await provider.send("eth_requestAccounts", []);
        }
        const network = await provider.getNetwork();
        if (network.chainId !== MONAD_CHAIN_ID) {
          try {
            await window.ethereum.request({
              method: 'wallet_switchEthereumChain',
              params: [{ chainId: '0x' + MONAD_CHAIN_ID.toString(16) }]
            });
          } catch (switchError) {
            if (switchError.code === 4902) {
              await window.ethereum.request({
                method: 'wallet_addEthereumChain',
                params: [{
                  chainId: '0x' + MONAD_CHAIN_ID.toString(16),
                  chainName: 'Monad Testnet',
                  nativeCurrency: { name: 'MON', symbol: 'MON', decimals: 18 },
                  rpcUrls: [RPC_URL],
                  blockExplorerUrls: [EXPLORER]
                }]
              });
            } else throw switchError;
          }
        }
        signer = provider.getSigner();
        account = await signer.getAddress();
        document.getElementById('walletStatus').textContent = account.slice(0,6) + '...' + account.slice(-4);
        document.getElementById('connectBtn').classList.add('hidden');
        document.getElementById('disconnectBtn').classList.remove('hidden');
        updateBalances();
        checkNFT();
      } catch (e) {
        console.error(e);
        if (!silent) alert("Connection failed: " + e.message);
      }
    }

    function disconnectWallet() {
      account = null;
      signer = null;
      document.getElementById('walletStatus').textContent = "Not connected";
      document.getElementById('connectBtn').classList.remove('hidden');
      document.getElementById('disconnectBtn').classList.add('hidden');
    }

    // Owner gate
    document.getElementById('unlockAdmin').onclick = () => {
      const key = document.getElementById('securityKey').value;
      if (key === OWNER_SECURITY_KEY) {
        isAdminUnlocked = true;
        document.getElementById('adminPanel').classList.remove('hidden');
        alert("Admin unlocked. Connect your owner wallet to perform actions.");
      } else {
        alert("Invalid security key.");
      }
    };

    document.getElementById('addCreatorBtn').onclick = async () => {
      if (!signer || !isAdminUnlocked) return alert("Unlock admin and connect wallet.");
      const registry = new ethers.Contract(CONTRACTS.registry, REGISTRY_ABI, signer);
      try {
        const tx = await registry.addCreator(
          document.getElementById('cName').value,
          document.getElementById('cBio').value,
          document.getElementById('cDesc').value,
          document.getElementById('cRemix').value,
          document.getElementById('cToken').value
        );
        await tx.wait();
        document.getElementById('adminStatus').innerHTML = '<div class="status success">Creator added!</div>';
        loadCreators();
      } catch (e) {
        document.getElementById('adminStatus').innerHTML = '<div class="status error">' + e.message + '</div>';
      }
    };

    document.getElementById('mintNFTBtn').onclick = async () => {
      if (!signer || !isAdminUnlocked) return alert("Unlock admin and connect wallet.");
      const nft = new ethers.Contract(CONTRACTS.tournamentNFT, NFT_ABI, signer);
      try {
        const tx = await nft.mint(document.getElementById('winnerAddr').value);
        await tx.wait();
        document.getElementById('adminStatus').innerHTML = '<div class="status success">NFT minted!</div>';
        if (account) checkNFT();
      } catch (e) {
        document.getElementById('adminStatus').innerHTML = '<div class="status error">' + e.message + '</div>';
      }
    };

    document.getElementById('addSwapTokenBtn').onclick = async () => {
      if (!signer || !isAdminUnlocked) return alert("Unlock admin and connect wallet.");
      const registry = new ethers.Contract(CONTRACTS.registry, REGISTRY_ABI, signer);
      try {
        const tx = await registry.addSwappableToken(document.getElementById('swapTokenAddr').value);
        await tx.wait();
        document.getElementById('adminStatus').innerHTML = '<div class="status success">Token added to swap!</div>';
        loadSwapTokens();
      } catch (e) {
        document.getElementById('adminStatus').innerHTML = '<div class="status error">' + e.message + '</div>';
      }
    };

    async function loadCreators() {
      if (CONTRACTS.registry === "0x0000000000000000000000000000000000000000") {
        document.getElementById('creatorsList').innerHTML = '<p>Deploy contracts and update addresses in the script.</p>';
        return;
      }
      try {
        const registry = new ethers.Contract(CONTRACTS.registry, REGISTRY_ABI, provider || new ethers.providers.JsonRpcProvider(RPC_URL));
        const count = await registry.creatorCount();
        let html = '';
        for (let i = 0; i < count; i++) {
          const c = await registry.getCreator(i);
          if (c.active) {
            html += `<div class="creator-card">
              <h3>${c.name}</h3>
              <p>${c.bio}</p>
              <p><small>${c.description}</small></p>
              <p>Remix: <a href="${c.remixProfile}" target="_blank">${c.remixProfile}</a></p>
              <p>Fan Token: ${c.fanToken.slice(0,6)}...</p>
            </div>`;
          }
        }
        document.getElementById('creatorsList').innerHTML = html || '<p>No creators yet.</p>';
      } catch (e) {
        document.getElementById('creatorsList').innerHTML = '<p class="error">Error loading creators. Check contract address.</p>';
      }
    }

    async function updateBalances() {
      if (!account || CONTRACTS.fanToken === "0x0000000000000000000000000000000000000000") return;
      try {
        const fan = new ethers.Contract(CONTRACTS.fanToken, ERC20_ABI, provider);
        const bal = await fan.balanceOf(account);
        document.getElementById('fanBalance').textContent = ethers.utils.formatUnits(bal, 18);
        const usdc = new ethers.Contract(CONTRACTS.mockUSDC, ERC20_ABI, provider);
        const usdcBal = await usdc.balanceOf(account);
        document.getElementById('usdcBalance').textContent = ethers.utils.formatUnits(usdcBal, 6);
      } catch (e) { console.error(e); }
    }

    document.getElementById('getFanToken').onclick = async () => {
      if (!signer) return alert("Connect wallet");
      const fan = new ethers.Contract(CONTRACTS.fanToken, ERC20_ABI, signer);
      try {
        const tx = await fan.faucet();
        await tx.wait();
        updateBalances();
        document.getElementById('tournamentStatus').innerHTML = '<div class="status success">100 Fan Tokens minted (demo)</div>';
      } catch (e) {
        document.getElementById('tournamentStatus').innerHTML = '<div class="status error">' + e.message + '</div>';
      }
    };

    document.getElementById('enterTournament').onclick = async () => {
      if (!account) return alert("Connect wallet");
      const fan = new ethers.Contract(CONTRACTS.fanToken, ERC20_ABI, provider);
      const bal = await fan.balanceOf(account);
      if (bal.gte(ethers.utils.parseUnits("1", 18))) {
        document.getElementById('tournamentStatus').innerHTML = '<div class="status success">You hold Fan Token — Tournament entered! (Demo: game would open here)</div>';
        // In real: window.open game link
      } else {
        document.getElementById('tournamentStatus').innerHTML = '<div class="status warning">Need at least 1 Fan Token to enter.</div>';
      }
    };

    async function checkNFT() {
      if (!account || CONTRACTS.tournamentNFT === "0x0000000000000000000000000000000000000000") return;
      try {
        const nft = new ethers.Contract(CONTRACTS.tournamentNFT, NFT_ABI, provider);
        const count = await nft.balanceOf(account);
        document.getElementById('nftCount').textContent = count.toString();
        document.getElementById('airdropEligible').innerHTML = count.gt(0)
          ? '<div class="status success">You hold ' + count + ' Tournament NFT(s) → Eligible for future project airdrop (not live yet)</div>'
          : '<div class="status warning">No Tournament NFT — not eligible yet</div>';
      } catch (e) { console.error(e); }
    }
    document.getElementById('checkNFT').onclick = checkNFT;

    document.getElementById('getUSDC').onclick = async () => {
      if (!signer) return alert("Connect wallet");
      const usdc = new ethers.Contract(CONTRACTS.mockUSDC, ERC20_ABI, signer);
      try {
        const tx = await usdc.faucet();
        await tx.wait();
        updateBalances();
      } catch (e) { alert(e.message); }
    };

    async function loadMarket() {
      if (CONTRACTS.predictionMarket === "0x0000000000000000000000000000000000000000") {
        document.getElementById('marketInfo').innerHTML = "Deploy PredictionMarket and update address.";
        return;
      }
      try {
        const market = new ethers.Contract(CONTRACTS.predictionMarket, MARKET_ABI, provider || new ethers.providers.JsonRpcProvider(RPC_URL));
        const info = await market.getMarketInfo();
        document.getElementById('marketInfo').innerHTML = `
          <strong>${info[0]}</strong><br>
          Option A: ${info[1]} (pool: ${ethers.utils.formatUnits(info[5], 6)})<br>
          Option B: ${info[2]} (pool: ${ethers.utils.formatUnits(info[6], 6)})<br>
          Settled: ${info[3]} ${info[3] ? '(Winner: ' + (info[4]===1?'A':'B') + ')' : ''}
        `;
      } catch (e) {
        document.getElementById('marketInfo').innerHTML = "Error loading market.";
      }
    }

    document.getElementById('betA').onclick = () => placeBet(1);
    document.getElementById('betB').onclick = () => placeBet(2);

    async function placeBet(option) {
      if (!signer) return alert("Connect wallet");
      const market = new ethers.Contract(CONTRACTS.predictionMarket, MARKET_ABI, signer);
      const usdc = new ethers.Contract(CONTRACTS.mockUSDC, ERC20_ABI, signer);
      const amount = ethers.utils.parseUnits("10", 6);
      try {
        await (await usdc.approve(CONTRACTS.predictionMarket, amount)).wait();
        const tx = await market.placeBet(option, amount);
        await tx.wait();
        document.getElementById('predStatus').innerHTML = '<div class="status success">Bet placed!</div>';
        loadMarket();
        updateBalances();
      } catch (e) {
        document.getElementById('predStatus').innerHTML = '<div class="status error">' + e.message + '</div>';
      }
    }

    async function loadSwapTokens() {
      if (CONTRACTS.registry === "0x0000000000000000000000000000000000000000") return;
      try {
        const registry = new ethers.Contract(CONTRACTS.registry, REGISTRY_ABI, provider || new ethers.providers.JsonRpcProvider(RPC_URL));
        const tokens = await registry.getSwappableTokens();
        if (tokens.length === 0) {
          document.getElementById('swapTokens').textContent = "No tokens added yet.";
          return;
        }
        document.getElementById('swapUI').classList.remove('hidden');
        let opts = tokens.map(t => `<option value="${t}">${t.slice(0,8)}...</option>`).join('');
        document.getElementById('swapFrom').innerHTML = opts;
        document.getElementById('swapTo').innerHTML = opts;
        document.getElementById('swapTokens').innerHTML = "Swappable tokens: " + tokens.map(t => t.slice(0,6)).join(', ');
      } catch (e) { console.error(e); }
    }

    document.getElementById('doSwap').onclick = async () => {
      if (!signer) return alert("Connect");
      const from = document.getElementById('swapFrom').value;
      const to = document.getElementById('swapTo').value;
      const amt = ethers.utils.parseUnits(document.getElementById('swapAmount').value || "1", 18);
      const token = new ethers.Contract(from, ERC20_ABI, signer);
      try {
        // Demo "swap": just transfer to self or a dummy (real would be AMM)
        const tx = await token.transfer(account, amt); // no-op-ish for demo
        await tx.wait();
        alert("Demo swap simulated (transfer to self). In production, integrate Uniswap or custom AMM.");
      } catch (e) { alert(e.message); }
    };

    // Initial loads
    setTimeout(loadSwapTokens, 500);
  </script>
